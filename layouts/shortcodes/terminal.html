{{/* Sesión de terminal: comando + salida, en una ventana con título. Uso: {{< terminal cmd="hpintel ip 1.2.3.4 --deep" title="reputación" max="20" >}} ...salida tal cual... {{< /terminal >}} - cmd: comando (opcional; sin él, solo salida). Se copia con el botón. - prompt: prompt decorativo (default `$`, no se copia). - title: texto de la barra de la ventana (opcional). - max: líneas visibles; el resto queda plegado en un
(sin JS). La salida va literal (sin markdown); el motor de plantillas la escapa. */}} {{- $prompt := (.Get "prompt") | default "$" -}} {{- $cmd := .Get "cmd" -}} {{- $title := .Get "title" -}} {{- $max := int ((.Get "max") | default "0") -}} {{- $out := trim .Inner "\n" -}} {{- $lines := split $out "\n" -}} {{- if and $cmd (not (.Page.Store.Get "hasCmd")) -}} {{- .Page.Store.Set "hasCmd" true -}} {{- end -}}
