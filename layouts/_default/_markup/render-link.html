{{- /* Enlaces markdown. Los externos (http/https a otro host) se abren en una pestaña nueva con rel="noopener noreferrer"; los internos (rutas relativas, anclas, el propio dominio) siguen en la misma pestaña. El destino se emite tal cual, igual que sin hook. */ -}} {{- $u := urls.Parse .Destination -}} {{- $site := urls.Parse site.BaseURL -}} {{- $external := and (in (slice "http" "https") $u.Scheme) (ne $u.Host $site.Host) -}} {{- .Text -}} {{- /* sin salto de línea final: el hook es inline */ -}}
