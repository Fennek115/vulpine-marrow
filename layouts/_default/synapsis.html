Clave = path.Base, igual que partials/wikilinks.html y backlinks.html. ============================================================ */}} {{- $pages := where .Site.RegularPages "Type" "posts" -}} {{- $byKey := dict -}} {{- range $pages -}} {{- $byKey = merge $byKey (dict (path.Base .RelPermalink) .) -}} {{- end -}} {{- $edges := slice -}} {{- $nodes := slice -}} {{- $seen := dict -}} {{- range $pages -}} {{- $src := path.Base .RelPermalink -}} {{- range findRE `\[\[[^\]]+\]\]` .RawContent -}} {{- $inner := replaceRE `^\[\[|\]\]$` "" . -}} {{- $tgt := trim (index (split $inner "|") 0) " " -}} {{- if and (isset $byKey $tgt) (ne $src $tgt) -}} {{- $edges = $edges | append (dict "s" $src "t" $tgt) -}} {{- range (slice $src $tgt) -}} {{- if not (isset $seen .) -}} {{- $seen = merge $seen (dict . true) -}} {{- $nodes = $nodes | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* ============================================================ Aristas automáticas — afinidad temática vía el motor Related de Hugo (tags/series, config [related] en hugo.toml).
