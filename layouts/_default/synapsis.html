{{/* Filtro por año — compartido por el grafo y el índice cronológico (js/synapsis.js). Hugo arma la definición mermaid desde los wikilinks [[ ]]; mermaid (dagre) la dibuja en capas. Clave = path.Base, igual que partials/wikilinks.html y backlinks.html. ============================================================ */}} {{- $pages := where .Site.RegularPages "Type" "posts" -}} {{- $byKey := dict -}} {{- range $pages -}} {{- $byKey = merge $byKey (dict (path.Base .RelPermalink) .) -}} {{- end -}} {{- $edges := slice -}} {{- $nodes := slice -}} {{- $seen := dict -}} {{- range $pages -}} {{- $src := path.Base .RelPermalink -}} {{- range findRE `\[\[[^\]]+\]\]` .RawContent -}} {{- $inner := replaceRE `^\[\[|\]\]$` "" . -}} {{- $tgt := trim (index (split $inner "|") 0) " " -}} {{- if and (isset $byKey $tgt) (ne $src $tgt) -}} {{- $edges = $edges | append (dict "s" $src "t" $tgt) -}} {{- range (slice $src $tgt) -}} {{- if not (isset $seen .) -}} {{- $seen = merge $seen (dict . true) -}} {{- $nodes = $nodes | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* ============================================================ Aristas automáticas — afinidad temática vía el motor Related de Hugo (tags/series, config [related] en hugo.toml). Solo el blog: la serie del Códice queda fuera (sus capítulos comparten tags/series y formarían un clúster que ahoga el grafo; ya tienen prev/next). Entran al grafo solo si un post los cita con [[ ]]. No se duplican pares ya unidos por wikilink explícito. ============================================================ */}} {{- $autoEdges := slice -}} {{- $pairSeen := dict -}} {{- range $e := $edges -}} {{- $pairSeen = merge $pairSeen (dict (delimit (sort (slice $e.s $e.t)) "->") true) -}} {{- end -}} {{- $blogPages := where $pages "Type" "posts" -}} {{- $blogPages = complement (where $blogPages "Params.series" "intersect" (slice $series)) $blogPages -}} {{- range $p := $blogPages -}} {{- $src := path.Base $p.RelPermalink -}} {{- $cands := where ($p.Site.RegularPages.Related $p) "Type" "posts" -}} {{- $cands = complement (where $cands "Params.series" "intersect" (slice $series)) $cands -}} {{- range first 3 $cands -}} {{- $tgt := path.Base .RelPermalink -}} {{- $pair := delimit (sort (slice $src $tgt)) "->" -}} {{- if not (isset $pairSeen $pair) -}} {{- $pairSeen = merge $pairSeen (dict $pair true) -}} {{- $autoEdges = $autoEdges | append (dict "s" $src "t" $tgt) -}} {{- range (slice $src $tgt) -}} {{- if not (isset $seen .) -}} {{- $seen = merge $seen (dict . true) -}} {{- $nodes = $nodes | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* key -> id secuencial; conteo de entradas (backlinks) para el hub */}} {{- $idmap := dict -}} {{- range $i, $k := $nodes -}}{{- $idmap = merge $idmap (dict $k (printf "n%d" $i)) -}}{{- end -}} {{- $incount := dict -}} {{- range $e := $edges -}} {{- $t := $e.t -}} {{- $incount = merge $incount (dict $t (add (int (index $incount $t | default 0)) 1)) -}} {{- end -}}
